cmake_minimum_required(VERSION 3.30.5)
project(NativeCourseExperiments LANGUAGES CXX)
if(NOT CMAKE_VERSION VERSION_EQUAL "3.30.5" OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the locked CMake 3.30.5 / MSVC x64 toolchain")
endif()
if(NOT CMAKE_CXX_COMPILER MATCHES "14[.]44[.]35207" OR NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Use MSVC 14.44.35207 with Debug or Release")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry features calib stereo video videoio dnn)
get_filename_component(CVLAB_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
find_package(spdlog 1.17.0 EXACT CONFIG REQUIRED PATHS "${CVLAB_ROOT}/build/deps/vcpkg_installed/cvlab-x64-windows-static-md/share/spdlog" NO_DEFAULT_PATH)
add_executable(native_labs NativeLabs.cpp "${CVLAB_ROOT}/src/app/ApplicationRuntime.cpp" "${CVLAB_ROOT}/src/core/diagnostics/Logger.cpp")
target_compile_features(native_labs PRIVATE cxx_std_20)
target_compile_options(native_labs PRIVATE /W4 /WX /utf-8 /permissive- /Zc:__cplusplus /EHsc)
target_include_directories(native_labs SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_include_directories(native_labs PRIVATE "${CVLAB_ROOT}/src")
target_compile_definitions(native_labs PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(native_labs PRIVATE ${OpenCV_LIBS} Qt6::Core spdlog::spdlog)
include(CTest)
foreach(topic IN ITEMS features camera video flow dnn)
    add_test(NAME native_${topic} COMMAND native_labs ${topic} "${CMAKE_BINARY_DIR}/output" "${CMAKE_BINARY_DIR}/relu.onnx")
    set_tests_properties(native_${topic} PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endforeach()
