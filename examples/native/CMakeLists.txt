cmake_minimum_required(VERSION 3.30.5)
project(NativeCourseExperiments LANGUAGES CXX)
if(NOT CMAKE_VERSION VERSION_EQUAL "3.30.5" OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use the locked CMake 3.30.5 / MSVC x64 toolchain")
endif()
if(NOT CMAKE_CXX_COMPILER MATCHES "14[.]44[.]35207" OR NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Use MSVC 14.44.35207 with Debug or Release")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry features calib stereo video videoio dnn)
get_filename_component(CVLAB_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Core)
find_package(spdlog 1.17.0 EXACT CONFIG REQUIRED PATHS "${CVLAB_ROOT}/build/deps/vcpkg_installed/cvlab-x64-windows-static-md/share/spdlog" NO_DEFAULT_PATH)
add_executable(native_labs NativeLabs.cpp "${CVLAB_ROOT}/src/app/ApplicationRuntime.cpp" "${CVLAB_ROOT}/src/core/diagnostics/Logger.cpp")
target_compile_features(native_labs PRIVATE cxx_std_20)
target_compile_options(native_labs PRIVATE /W4 /WX /utf-8 /permissive- /Zc:__cplusplus /EHsc)
target_include_directories(native_labs SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_include_directories(native_labs PRIVATE "${CVLAB_ROOT}/src")
target_compile_definitions(native_labs PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(native_labs PRIVATE ${OpenCV_LIBS} Qt6::Core spdlog::spdlog)
include(CTest)
foreach(topic IN ITEMS features camera video flow dnn)
    add_test(NAME native_${topic} COMMAND native_labs ${topic} "${CMAKE_BINARY_DIR}/output" "${CMAKE_BINARY_DIR}/relu.onnx")
    set_tests_properties(native_${topic} PROPERTIES TIMEOUT 60
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endforeach()
option(CVLAB_WITH_MODEL "Enable downloaded SqueezeNet test" OFF)
foreach(lab IN ITEMS real_image_labs real_model_lab)
    if(lab STREQUAL "real_image_labs")
        set(lab_source RealImageLabs.cpp)
    else()
        set(lab_source RealModelLab.cpp)
    endif()
    add_executable(${lab} ${lab_source} "${CVLAB_ROOT}/src/app/ApplicationRuntime.cpp" "${CVLAB_ROOT}/src/core/diagnostics/Logger.cpp")
    target_compile_features(${lab} PRIVATE cxx_std_20)
    target_compile_options(${lab} PRIVATE /W4 /WX /utf-8 /permissive- /Zc:__cplusplus /EHsc)
    target_include_directories(${lab} SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
    target_include_directories(${lab} PRIVATE "${CVLAB_ROOT}/src")
    target_compile_definitions(${lab} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_link_libraries(${lab} PRIVATE ${OpenCV_LIBS} Qt6::Core spdlog::spdlog)
endforeach()
foreach(topic IN ITEMS image registration video inspection)
    add_test(NAME real_${topic} COMMAND real_image_labs ${topic} "${CVLAB_ROOT}/tests/fixtures/real" "${CMAKE_BINARY_DIR}/real-output")
    set_tests_properties(real_${topic} PROPERTIES TIMEOUT 60 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endforeach()
add_test(NAME real_missing_image COMMAND real_image_labs image "${CVLAB_ROOT}/tests/fixtures/absent" "${CMAKE_BINARY_DIR}/real-output")
add_test(NAME real_invalid_mode COMMAND real_image_labs invalid "${CVLAB_ROOT}/tests/fixtures/real" "${CMAKE_BINARY_DIR}/real-output")
set_tests_properties(real_missing_image real_invalid_mode PROPERTIES WILL_FAIL TRUE TIMEOUT 30
    ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
if(CVLAB_WITH_MODEL)
    add_test(NAME real_video_ai COMMAND real_image_labs videoai "${CVLAB_ROOT}/tests/fixtures/real" "${CMAKE_BINARY_DIR}/real-output" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx")
    set_tests_properties(real_video_ai PROPERTIES TIMEOUT 120 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    add_test(NAME real_video_ai_missing_model COMMAND real_image_labs videoai "${CVLAB_ROOT}/tests/fixtures/real" "${CMAKE_BINARY_DIR}/real-output" "${CVLAB_ROOT}/build/models/missing.onnx")
    set_tests_properties(real_video_ai_missing_model PROPERTIES WILL_FAIL TRUE TIMEOUT 30
        ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    add_test(NAME real_model COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/astronaut.png" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx" --verify-fixture --labels "${CVLAB_ROOT}/tests/fixtures/real/imagenet-synset.txt")
    add_test(NAME real_model_missing_labels COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/astronaut.png" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx" --labels "${CVLAB_ROOT}/tests/fixtures/real/missing-labels.txt")
    add_test(NAME real_model_other_image COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/cell.png" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx")
    add_test(NAME real_model_fixture_mismatch COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/cell.png" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx" --verify-fixture)
    add_test(NAME real_model_missing_image COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/missing.png" "${CVLAB_ROOT}/build/models/squeezenet1.1-7.onnx")
    add_test(NAME real_model_missing_model COMMAND real_model_lab "${CVLAB_ROOT}/tests/fixtures/real/astronaut.png" "${CVLAB_ROOT}/build/models/missing.onnx")
    set_tests_properties(real_model_missing_image real_model_missing_model real_model_fixture_mismatch real_model_missing_labels PROPERTIES WILL_FAIL TRUE TIMEOUT 60 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
    set_tests_properties(real_model real_model_other_image PROPERTIES TIMEOUT 60 ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:$<TARGET_FILE_DIR:opencv_core>;PATH=path_list_prepend:$<TARGET_FILE_DIR:Qt6::Core>")
endif()
