cmake_minimum_required(VERSION 3.30.5)
project(WorkbenchReference LANGUAGES CXX)
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use locked MSVC x64 environment")
endif()
if(NOT CMAKE_CXX_COMPILER MATCHES "14[.]44[.]35207")
    message(FATAL_ERROR "Use locked MSVC 14.44.35207 compiler")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR)
    set(CMAKE_SYSTEM_PROCESSOR AMD64)
endif()
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry)
add_executable(workbench_reference WIN32 main.cpp)
target_compile_features(workbench_reference PRIVATE cxx_std_20)
target_compile_options(workbench_reference PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
target_include_directories(workbench_reference SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(workbench_reference PRIVATE Qt6::Widgets opencv_core opencv_imgproc opencv_imgcodecs opencv_geometry)
