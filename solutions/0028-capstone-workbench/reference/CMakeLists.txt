cmake_minimum_required(VERSION 3.30.5)
project(WorkbenchReference LANGUAGES CXX)
if(NOT CMAKE_VERSION VERSION_EQUAL "3.30.5")
    message(FATAL_ERROR "Use locked CMake 3.30.5")
endif()
if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Use locked MSVC x64 environment")
endif()
if(NOT CMAKE_CXX_COMPILER MATCHES "14[.]44[.]35207")
    message(FATAL_ERROR "Use locked MSVC 14.44.35207 compiler")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR)
    set(CMAKE_SYSTEM_PROCESSOR AMD64)
endif()
if(NOT CMAKE_BUILD_TYPE MATCHES "^(Debug|Release)$")
    message(FATAL_ERROR "Select Debug or Release")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets)
find_package(OpenCV 5.0.0 EXACT REQUIRED COMPONENTS core imgproc imgcodecs geometry)
get_filename_component(CVLAB_ROOT "../../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
find_package(spdlog 1.17.0 EXACT CONFIG REQUIRED PATHS "${CVLAB_ROOT}/build/deps/vcpkg_installed/cvlab-x64-windows-static-md/share/spdlog" NO_DEFAULT_PATH)
add_executable(workbench_reference WIN32 main.cpp workbench.ui resources.qrc workbench.qss
    business/Processing.h business/Processing.cpp infrastructure/ParametersFile.h infrastructure/ParametersFile.cpp
    "${CVLAB_ROOT}/src/app/ApplicationRuntime.cpp" "${CVLAB_ROOT}/src/core/diagnostics/Logger.cpp")
set_target_properties(workbench_reference PROPERTIES AUTOUIC ON AUTORCC ON)
target_compile_features(workbench_reference PRIVATE cxx_std_20)
target_compile_options(workbench_reference PRIVATE /W4 /WX /permissive- /utf-8 /EHsc)
target_include_directories(workbench_reference SYSTEM PRIVATE ${OpenCV_INCLUDE_DIRS})
target_include_directories(workbench_reference PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}" "${CVLAB_ROOT}/src")
target_compile_definitions(workbench_reference PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(workbench_reference PRIVATE Qt6::Widgets opencv_core opencv_imgproc opencv_imgcodecs opencv_geometry spdlog::spdlog)
add_library(workbench_header_checks OBJECT tests/HeaderProcessing.cpp tests/HeaderParametersFile.cpp)
target_compile_features(workbench_header_checks PRIVATE cxx_std_20)
target_compile_options(workbench_header_checks PRIVATE /utf-8)
target_include_directories(workbench_header_checks PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(workbench_header_checks PRIVATE opencv_core)
